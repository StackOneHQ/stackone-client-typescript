## Example Usage

```typescript
import { ActionsRpcSyncedRequestDto } from "@stackone/stackone-client-ts/sdk/models/shared";

let value: ActionsRpcSyncedRequestDto = {
  action: "create_employee",
  aggs: {
    "by_status": {
      "terms": {
        "field": "employment_status",
      },
    },
  },
  filter: {
    "match": {
      "description": "onboarding",
    },
  },
  search: "onboarding safety",
  sort: [
    {
      "last_name": "desc",
    },
  ],
  source: [
    "first_name",
    "email",
  ],
  syncId: "sch_01J8Z3Q4",
};
```
